# Sicurezza - luxembourgfestival.com (hosting Linux Aruba / Apache)
# Caricare nella cartella principale del sito, accanto a index.html.

# Forza HTTPS
<IfModule mod_rewrite.c>
  RewriteEngine On
  RewriteCond %{HTTPS} !=on
  RewriteCond %{HTTP:X-Forwarded-Proto} !=https
  RewriteRule ^(.*)$ https://%{HTTP_HOST}%{REQUEST_URI} [L,R=301]

  # Blocca file nascosti (tranne .well-known) e file di backup/sorgenti
  RewriteRule ^\.(?!well-known/) - [F,L]
  RewriteRule \.(bak|old|orig|swp|sql|log|ini|sh|md)$ - [F,L,NC]
</IfModule>

# Niente elenco dei file nelle cartelle
<IfModule mod_autoindex.c>
  IndexIgnore *
</IfModule>

# Header di sicurezza
<IfModule mod_headers.c>
  Header always set Strict-Transport-Security "max-age=15552000; includeSubDomains"
  Header always set X-Content-Type-Options "nosniff"
  Header always set X-Frame-Options "SAMEORIGIN"
  Header always set Referrer-Policy "strict-origin-when-cross-origin"
  Header always set Permissions-Policy "camera=(), microphone=(), geolocation=(), payment=(), usb=()"
  Header always set Cross-Origin-Opener-Policy "same-origin-allow-popups"
  Header always set Content-Security-Policy "default-src 'self'; script-src 'self' 'unsafe-inline' https://www.googletagmanager.com https://connect.facebook.net; style-src 'self' 'unsafe-inline' https://fonts.googleapis.com; font-src 'self' https://fonts.gstatic.com; img-src 'self' data: https:; media-src 'self'; frame-src https://www.openstreetmap.org; connect-src 'self' https://*.google-analytics.com https://*.analytics.google.com https://www.googletagmanager.com https://www.facebook.com https://connect.facebook.net; form-action 'self'; base-uri 'self'; object-src 'none'; frame-ancestors 'self'"
  Header unset X-Powered-By
</IfModule>

# Nasconde la versione di PHP e proteggendo send.php da usi diversi da POST
<IfModule mod_rewrite.c>
  RewriteCond %{REQUEST_URI} ^/send\.php$ [NC]
  RewriteCond %{REQUEST_METHOD} !^(POST|OPTIONS)$ [NC]
  RewriteRule ^ - [F,L]
</IfModule>
